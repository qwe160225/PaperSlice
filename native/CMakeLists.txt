cmake_minimum_required(VERSION 3.16)
project(paperslice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(USE_IMPLICIT_CRYPTO OFF CACHE BOOL "" FORCE)
set(REQUIRE_CRYPTO_NATIVE ON CACHE BOOL "" FORCE)
set(STATIC_JPEG ON CACHE BOOL "" FORCE)
set(BUILD_DOC OFF CACHE BOOL "" FORCE)
set(INSTALL_MANUAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../third_party/qpdf-12.4.2 qpdf EXCLUDE_FROM_ALL)
set_target_properties(libqpdf_object PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    add_library(paperslice SHARED core.cc jni.cc)
    target_link_options(paperslice PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
else()
    add_executable(paperslice-cli core.cc cli.cc)
    target_link_libraries(paperslice-cli PRIVATE libqpdf)
    find_package(JNI)
    if(JNI_FOUND)
        add_library(paperslice SHARED core.cc jni.cc)
        target_include_directories(paperslice PRIVATE ${JNI_INCLUDE_DIRS})
    endif()
endif()
if(TARGET paperslice)
    target_link_libraries(paperslice PRIVATE libqpdf)
endif()
